class PerspectiveGridPainter extends CustomPainter {
  @override
  void paint(Canvas canvas, Size size) {
    final paint = Paint()
      ..color = Colors.white.withValues(alpha: 0.12)
      ..strokeWidth = 1.2
      ..style = PaintingStyle.stroke;

    final double cx = size.width / 2;
    final double h = size.height;
    final double w = size.width;
    final Offset vp = Offset(cx, -h * 0.8);

    canvas.drawLine(vp, Offset(cx, h), paint);
    canvas.drawLine(vp, Offset(cx - w * 0.35, h), paint);
    canvas.drawLine(vp, Offset(cx + w * 0.35, h), paint);
    canvas.drawLine(vp, Offset(cx - w * 0.85, h), paint);
    canvas.drawLine(vp, Offset(cx + w * 0.85, h), paint);

    List<double> yStops = [0.2, 0.4, 0.65, 0.85, 1.0];
    for (double stop in yStops) {
      canvas.drawLine(Offset(0, h * stop), Offset(w, h * stop), paint);
    }
  }

  @override
  bool shouldRepaint(covariant CustomPainter oldDelegate) => false;
}

class GlassyGooeyNav extends StatelessWidget {
  final int currentIndex;
  final ValueChanged<int> onTap;

  GlassyGooeyNav({super.key, required this.currentIndex, required this.onTap});

  final List<IconData> lightIcons = [IconlyLight.home, IconlyLight.search, IconlyLight.category, IconlyLight.heart];
  final List<IconData> boldIcons = [IconlyBold.home, IconlyBold.search, IconlyBold.category, IconlyBold.heart];

  @override
  Widget build(BuildContext context) {
    return LayoutBuilder(
      builder: (context, constraints) {
        final double width = constraints.maxWidth;
        const double navHeight = 75.0;
        const double horizontalPadding = 20.0;
        final double itemWidth = (width - (horizontalPadding * 2)) / 4;
        final double targetX = horizontalPadding + (currentIndex * itemWidth) + (itemWidth / 2);

        return ClipRRect(
          borderRadius: const BorderRadius.vertical(top: Radius.circular(30)),
          child: BackdropFilter(
            filter: ImageFilter.blur(sigmaX: 20, sigmaY: 20),
            child: Container(
              height: navHeight,
              decoration: BoxDecoration(
                color: Colors.white.withValues(alpha: 0.08),
                border: Border(top: BorderSide(color: Colors.white.withValues(alpha: 0.1), width: 1)),
              ),
              child: Stack(
                children: [
                  TweenAnimationBuilder<double>(
                    tween: Tween<double>(end: targetX),
                    duration: const Duration(milliseconds: 350),
                    curve: Curves.easeOutCubic,
                    builder: (context, xOffset, child) {
                      return CustomPaint(painter: DropletPainter(xOffset: xOffset), size: Size(width, navHeight));
                    },
                  ),
                  Padding(
                    padding: const EdgeInsets.symmetric(horizontal: horizontalPadding),
                    child: Row(
                      children: List.generate(4, (index) {
                        final isSelected = index == currentIndex;
                        return GestureDetector(
                          onTap: () => onTap(index),
                          behavior: HitTestBehavior.opaque,
                          child: SizedBox(
                            width: itemWidth,
                            height: navHeight,
                            child: Center(
                              child: AnimatedContainer(
                                duration: const Duration(milliseconds: 350),
                                transform: Matrix4.translationValues(0, isSelected ? -1.5 : 0, 0),
                                child: Icon(
                                  isSelected ? boldIcons[index] : lightIcons[index],
                                  color: isSelected ? Colors.white : Colors.white54,
                                  size: 26,
                                ),
                              ),
                            ),
                          ),
                        );
                      }),
                    ),
                  ),
                ],
              ),
            ),
          ),
        );
      },
    );
  }
}

class DropletPainter extends CustomPainter {
  final double xOffset;
  DropletPainter({required this.xOffset});

  @override
  void paint(Canvas canvas, Size size) {
    final paint = Paint()
      ..color = const Color(0xFF050505)
      ..style = PaintingStyle.fill;
    final path = Path();
    double cx = xOffset;
    const double radius = 24.0;
    const double centerY = 32.0;

    path.moveTo(cx - 40, 0);
    path.cubicTo(cx - 4, 0, cx - radius, 20, cx - radius, centerY);
    path.arcTo(Rect.fromCircle(center: Offset(cx, centerY), radius: radius), math.pi, -math.pi, false);
    path.cubicTo(cx + radius, 20, cx + 4, 0, cx + 40, 0);
    path.close();

    canvas.drawPath(path, paint);
  }

  @override
  bool shouldRepaint(covariant DropletPainter oldDelegate) => oldDelegate.xOffset != xOffset;
}
