class CheckerboardPainter extends CustomPainter {
  @override
  void paint(Canvas canvas, Size size) {
    final paint = Paint()..color = Colors.white.withValues(alpha: 0.5);
    final double squareSize = size.width / 8;
    for (int i = 0; i < 8; i++) {
      for (int j = 0; j < 5; j++) { 
        if ((i + j) % 2 == 0) {
          final double opacity = 1.0 - (j / 5.0);
          paint.color = Colors.white.withValues(alpha: 0.5 * opacity);
          canvas.drawRect(Rect.fromLTWH(i * squareSize, j * squareSize, squareSize, squareSize), paint);
        }
      }
    }
  }
  @override
  bool shouldRepaint(covariant CustomPainter oldDelegate) => false;
}

class LightGlassyGooeyNav extends StatelessWidget {
  final int currentIndex;
  final ValueChanged<int> onTap;

  LightGlassyGooeyNav({super.key, required this.currentIndex, required this.onTap});

  final List<IconData> lightIcons = [IconlyLight.home, IconlyLight.search, Icons.leaderboard_outlined, IconlyLight.setting];
  final List<IconData> boldIcons = [IconlyBold.home, IconlyBold.search, Icons.leaderboard, IconlyBold.setting];

  @override
  Widget build(BuildContext context) {
    return LayoutBuilder(
      builder: (context, constraints) {
        final double width = constraints.maxWidth;
        const double navHeight = 85.0;
        const double horizontalPadding = 20.0;
        final double itemWidth = (width - (horizontalPadding * 2)) / 4;
        final double targetX = horizontalPadding + (currentIndex * itemWidth) + (itemWidth / 2);

        return ClipRRect(
          borderRadius: const BorderRadius.vertical(top: Radius.circular(40)),
          child: BackdropFilter(
            filter: ImageFilter.blur(sigmaX: 25, sigmaY: 25),
            child: Container(
              height: navHeight,
              decoration: BoxDecoration(
                color: Colors.white.withValues(alpha: 0.7), 
                border: Border(top: BorderSide(color: Colors.white.withValues(alpha: 0.9), width: 1.5)),
              ),
              child: Stack(
                children: [
                  TweenAnimationBuilder<double>(
                    tween: Tween<double>(end: targetX),
                    duration: const Duration(milliseconds: 350),
                    curve: Curves.easeOutCubic,
                    builder: (context, xOffset, child) {
                      return CustomPaint(painter: WhiteDropletPainter(xOffset: xOffset), size: Size(width, navHeight));
                    },
                  ),
                  Padding(
                    padding: const EdgeInsets.symmetric(horizontal: horizontalPadding),
                    child: Row(
                      children: List.generate(4, (index) {
                        final isSelected = index == currentIndex;
                        return GestureDetector(
                          onTap: () => onTap(index),
                          behavior: HitTestBehavior.opaque,
                          child: SizedBox(
                            width: itemWidth,
                            height: navHeight,
                            child: Center(
                              child: AnimatedContainer(
                                duration: const Duration(milliseconds: 350),
                                transform: Matrix4.translationValues(0, isSelected ? -4.0 : 0, 0),
                                child: Icon(
                                  isSelected ? boldIcons[index] : lightIcons[index],
                                  color: isSelected ? const Color(0xFF2A5265) : Colors.grey.shade500,
                                  size: 28,
                                ),
                              ),
                            ),
                          ),
                        );
                      }),
                    ),
                  ),
                ],
              ),
            ),
          ),
        );
      },
    );
  }
}

class WhiteDropletPainter extends CustomPainter {
  final double xOffset;
  WhiteDropletPainter({required this.xOffset});

  @override
  void paint(Canvas canvas, Size size) {
    final paint = Paint()
      ..color = Colors.white 
      ..style = PaintingStyle.fill;
    
    final shadowPaint = Paint()
      ..color = Colors.black.withValues(alpha: 0.08)
      ..maskFilter = const MaskFilter.blur(BlurStyle.normal, 5);

    final path = Path();
    double cx = xOffset;
    const double radius = 26.0;
    const double centerY = 36.0;

    path.moveTo(cx - 50, 0);
    path.cubicTo(cx - 20, 0, cx - 22, 20, cx - radius, centerY);
    path.arcTo(Rect.fromCircle(center: Offset(cx, centerY), radius: radius), math.pi, -math.pi, false);
    path.cubicTo(cx + 22, 20, cx + 20, 0, cx + 50, 0);
    path.close();

    canvas.drawPath(path, shadowPaint); 
    canvas.drawPath(path, paint);       
  }

  @override
  bool shouldRepaint(covariant WhiteDropletPainter oldDelegate) => oldDelegate.xOffset != xOffset;
}
